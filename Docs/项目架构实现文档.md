-   心跳机制

# 3. EventBus

## event/EventBus.ts

``` ts
type Handler = (data?: any) => void

export class EventBus {

  private events: Map<string, Handler[]> = new Map()

  on(event: string, handler: Handler) {
    const handlers = this.events.get(event) || []
    handlers.push(handler)
    this.events.set(event, handlers)
  }

  off(event: string, handler: Handler) {
    const handlers = this.events.get(event)
    if (!handlers) return
    this.events.set(
      event,
      handlers.filter(h => h !== handler)
    )
  }

  emit(event: string, data?: any) {
    const handlers = this.events.get(event)
    if (!handlers) return
    handlers.forEach(h => h(data))
  }
}
```

# 5. MessageDispatcher

## core/MessageDispatcher.ts

``` ts
import { SDKResponse } from "../protocol/types"

interface PendingRequest {
  resolve: Function
  reject: Function
  timeout: any
}

export class MessageDispatcher {

  private pending = new Map<string, PendingRequest>()

  register(requestId: string) {
    return new Promise((resolve, reject) => {

      const timeout = setTimeout(() => {
        this.pending.delete(requestId)
        reject(new Error("Request timeout"))
      }, 20000)

      this.pending.set(requestId, { resolve, reject, timeout })

    })
  }

  resolve(response: SDKResponse) {
    const pending = this.pending.get(response.requestId)
    if (!pending) return

    clearTimeout(pending.timeout)
    pending.resolve(response)
    this.pending.delete(response.requestId)
  }

}
```

# 6. ConnectionManager

## core/ConnectionManager.ts

``` ts
import { EventBus } from "../event/EventBus"

export class ConnectionManager {

  private ws?: WebSocket
  private url: string
  private eventBus: EventBus

  constructor(url: string, eventBus: EventBus) {
    this.url = url
    this.eventBus = eventBus
  }

  connect() {

    this.ws = new WebSocket(this.url)

    this.ws.onopen = () => {
      this.eventBus.emit("open")
    }

    this.ws.onmessage = (event) => {
      this.eventBus.emit("message", event.data)
    }

    this.ws.onerror = (err) => {
      this.eventBus.emit("error", err)
    }

    this.ws.onclose = () => {
      this.eventBus.emit("close")
    }

  }

  send(data: any) {
    this.ws?.send(JSON.stringify(data))
  }

  close() {
    this.ws?.close()
  }

}
```

  private attempts = 0
  private maxAttempts = 5

  async wait() {

    const delay = Math.pow(2, this.attempts) * 1000
    this.attempts++

    return new Promise(resolve =>
      setTimeout(resolve, delay)
    )
  }

  reset() {
    this.attempts = 0

# 10. AgentSDK 主入口

## core/AgentSDK.ts

``` ts
import { EventBus } from "../event/EventBus"
import { ConnectionManager } from "./ConnectionManager"
import { MessageDispatcher } from "./MessageDispatcher"
import { SessionManager } from "./SessionManager"
import { uuid } from "../utils/uuid"
import { SDKOptions, SDKRequest } from "../protocol/types"

export class AgentSDK {

  private options: SDKOptions
  private connection: ConnectionManager
  private dispatcher = new MessageDispatcher()
  private eventBus = new EventBus()
  private sessionManager = new SessionManager()

  constructor(options: SDKOptions) {

    this.options = options

    this.connection = new ConnectionManager(
      options.url,
      this.eventBus
    )

    this.setupListeners()
  }

  private setupListeners() {

    this.eventBus.on("message", (data) => {

      const msg = JSON.parse(data)

      if (msg.type === "stream") {
        this.eventBus.emit("stream", msg)
      }

      if (msg.requestId) {
        this.dispatcher.resolve(msg)
      }

    })

  }

  connect() {
    this.connection.connect()
  }

  send(request: SDKRequest) {

    const requestId = uuid()
    request.requestId = requestId

    const promise = this.dispatcher.register(requestId)

    this.connection.send(request)

    return promise
  }

  on(event: string, handler: any) {
    this.eventBus.on(event, handler)
  }

}
```
